# 契约权限重构实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建立后端共享权限契约，并逐步让后端注解和前端按钮权限使用同一套权限码。

**Architecture:** 第一阶段先在 `auth-feignClient` 提供权限枚举、权限目录和兼容映射，不直接改变 controller 行为。随后后端注解改用常量，最后前端引入同名权限类型和映射，去掉纯字符串散落。

**Tech Stack:** JDK 21、Maven、JUnit 5、Vue 3、TypeScript、Vite。

## Global Constraints

- 文件路径只记录仓库内相对路径。
- 每完成一个任务提交一次。
- 生产代码变更前必须先写失败测试并确认失败。
- 后端 HTTP 响应继续遵循现有 `RI<T>` 结构。
- 权限码目标格式采用三段式：`system:user:create`、`system:menu:view`、`education:course:view`。
- 迁移期保留旧权限码兼容映射，避免一次性改坏存量数据和菜单配置。
- 不在本阶段重写登录、菜单、角色分配业务流程。

```java
package com.xiwen.auth.client.registry;

import com.xiwen.auth.client.enums.PermissionCode;
import org.junit.jupiter.api.Test;

import java.util.List;
import java.util.Set;
import java.util.stream.Collectors;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertFalse;
import static org.junit.jupiter.api.Assertions.assertTrue;

class PermissionCatalogTest {

    @Test
    void shouldExposeEveryPermissionCodeOnce() {
        Set<String> catalogCodes = PermissionCatalog.all().stream()
                .map(descriptor -> descriptor.code().code())
                .collect(Collectors.toSet());

        assertEquals(PermissionCode.values().length, catalogCodes.size());
        assertTrue(catalogCodes.contains(PermissionCode.SYSTEM_MENU_VIEW.CODE));
        assertTrue(catalogCodes.contains(PermissionCode.SYSTEM_ROLE_ASSIGN.CODE));
        assertTrue(catalogCodes.contains(PermissionCode.EDUCATION_CLASS_ENROLL.CODE));
    }

    @Test
    void shouldNormalizeLegacyBackendPermissionCodes() {
        assertEquals(PermissionCode.SYSTEM_MENU_VIEW.CODE, PermissionCatalog.normalize("menu:view"));
        assertEquals(PermissionCode.SYSTEM_MENU_CREATE.CODE, PermissionCatalog.normalize("menu:create"));
        assertEquals(PermissionCode.EDUCATION_COURSE_DELETE.CODE, PermissionCatalog.normalize("course:delete"));
        assertEquals(PermissionCode.EDUCATION_CLASS_ENROLL.CODE, PermissionCatalog.normalize("class:enroll"));
        assertEquals(PermissionCode.EDUCATION_STUDENT_VIEW.CODE, PermissionCatalog.normalize("student:view"));
    }

    @Test
    void shouldKeepCanonicalPermissionCodesStable() {
        assertEquals(PermissionCode.SYSTEM_USER_CREATE.CODE, PermissionCatalog.normalize("system:user:create"));
        assertEquals(PermissionCode.SYSTEM_ROLE_ASSIGN.CODE, PermissionCatalog.normalize("system:role:assign"));
        assertEquals(PermissionCode.EDUCATION_COURSE_VIEW.CODE, PermissionCatalog.normalize("education:course:view"));
    }

    @Test
    void shouldReturnUnknownPermissionCodeWithoutLosingSignal() {
        assertEquals("custom:report:view", PermissionCatalog.normalize("custom:report:view"));
        assertFalse(PermissionCatalog.findByCode("custom:report:view").isPresent());
    }

    @Test
    void shouldExposeLegacyCodesFromDescriptor() {
        List<String> legacyCodes = PermissionCatalog.findByCode(PermissionCode.SYSTEM_MENU_VIEW.CODE)
                .orElseThrow()
                .legacyCodes();

        assertTrue(legacyCodes.contains("menu:view"));
    }
}
```

package com.xiwen.server.admin.controller;

import com.xiwen.authz.annotation.RequiresPermission;
import org.junit.jupiter.api.Test;

import java.lang.reflect.Method;
import java.util.Arrays;
import java.util.Set;
import java.util.stream.Collectors;

import static org.junit.jupiter.api.Assertions.assertFalse;
import static org.junit.jupiter.api.Assertions.assertTrue;

class AdminPermissionAnnotationTest {

    @Test
    void adminControllersShouldUseCanonicalSystemPermissionCodes() {
        Set<String> permissions = Arrays.stream(new Class<?>[]{
                        AdminMenuController.class,
                        AdminUserController.class,
                        AdminRoleController.class,
                        AdminRolePermissionController.class,
                        AdminUserRoleController.class
                })
                .flatMap(controller -> Arrays.stream(controller.getDeclaredMethods()))
                .map(Method::getDeclaredAnnotations)
                .flatMap(Arrays::stream)
                .filter(annotation -> annotation instanceof RequiresPermission)
                .map(annotation -> (RequiresPermission) annotation)
                .flatMap(annotation -> Arrays.stream(annotation.value()))
                .collect(Collectors.toSet());

        assertFalse(permissions.contains("menu:view"));
        assertFalse(permissions.contains("menu:create"));
        assertTrue(permissions.contains("system:menu:view"));
        assertTrue(permissions.contains("system:role:assign"));
    }
}
